1""" 

2Given a partially filled 9×9 2D array, the objective is to fill a 9×9 

3square grid with digits numbered 1 to 9, so that every row, column, and 

4and each of the nine 3×3 sub-grids contains all of the digits. 

5 

6This can be solved using Backtracking and is similar to n-queens. 

7We check to see if a cell is safe or not and recursively call the 

8function on the next column to see if it returns True. if yes, we 

9have solved the puzzle. else, we backtrack and place another number 

10in that cell and repeat this process. 

11"""

43def is_safe(grid: Matrix, row: int, column: int, n: int) -> bool: 

44 """ 

45 This function checks the grid to see if each row, 

46 column, and the 3x3 subgrids contain the digit 'n'. 

47 It returns False if it is not 'safe' (a duplicate digit 

48 is found) else returns True if it is 'safe' 

49 """ 

50 for i in range(9): 

51 if n in {grid[row][i], grid[i][column]}: 

52 return False 

53 

54 for i in range(3): 

55 for j in range(3): 

56 if grid[(row - row % 3) + i][(column - column % 3) + j] == n: 

57 return False 

58 

59 return True

74def sudoku(grid: Matrix) -> Matrix | None: 

75 """ 

76 Takes a partially filled-in grid and attempts to assign values to 

77 all unassigned locations in such a way to meet the requirements 

78 for Sudoku solution (non-duplication across rows, columns, and boxes) 

79 

80 >>> sudoku(initial_grid) # doctest: +NORMALIZE_WHITESPACE 

81 [[3, 1, 6, 5, 7, 8, 4, 9, 2], 

82 [5, 2, 9, 1, 3, 4, 7, 6, 8], 

83 [4, 8, 7, 6, 2, 9, 5, 3, 1], 

84 [2, 6, 3, 4, 1, 5, 9, 8, 7], 

85 [9, 7, 4, 8, 6, 3, 1, 2, 5], 

86 [8, 5, 1, 7, 9, 2, 6, 4, 3], 

87 [1, 3, 8, 9, 4, 7, 2, 5, 6], 

88 [6, 9, 2, 3, 5, 1, 8, 7, 4], 

89 [7, 4, 5, 2, 8, 6, 3, 1, 9]] 

90 >>> sudoku(no_solution) is None 

91 True 

92 """ 

93 if location := find_empty_location(grid): 

94 row, column = location 

95 else: 

96 # If the location is ``None``, then the grid is solved. 

97 return grid 

98 

99 for digit in range(1, 10): 

100 if is_safe(grid, row, column, digit): 

101 grid[row][column] = digit 

102 

103 if sudoku(grid) is not None: 

104 return grid 

105 

106 grid[row][column] = 0 

107 

108 return None
